calculate_svymean <- function(age_range, .svy) {
  .svy <- subset(
    .svy,
    between(RIDAGEYR, age_range, age_range + 4)
  )

  svyby(
    ~ BPXSY1 + BPXDI1,
    ~ noplace + female,
    .svy,
    svymean,
    na.rm = TRUE
  ) |>
    # wrangle the dataset to long format
    # for better plotting of sex vs bp type over age range
    as_tibble() |>
    mutate(
      noplace = factor(noplace, labels = c("No", "Yes")),
      female = factor(female, labels = c("Male", "Female")),
      age_range = paste0(age_range, "-", age_range + 4)
    ) |>
    select(age_range, noplace, female, systolic = BPXSY1, diastolic = BPXDI1) |>
    pivot_longer(
      c(systolic, diastolic),
      names_to = "bp_type",
      values_to = "bp"
    )
}

nhanes_no_treat <- nhanes |>

folds <- vfold_cv(nhanes_no_treat, v = 5)
splits <- vfold_cv(nhanes_no_treat, v = 5, strata = SDMVSTRA) # stratify if possible
results_df <- data.frame(
  iteration = integer(5), # integer column
  mse_val = numeric(5),
  rmse_val = numeric(5),
  mae_val = numeric(5) # numeric column
)
column_list <- c(
  "RIDAGEYR", "female", "noplace", "poor", "education_L12", "singlefamily", "crowded", "rented", "notemployed", "BPXSY1",
  "BPXDI1"
)
for (i in 1:5) {
  # Split the data
  train_data <- analysis(folds$splits[[i]])
  test_data <- assessment(folds$splits[[i]])

  clean_test_data <- test_data[complete.cases(test_data[, column_list]), ]

  train_design <- svydesign(id = ~SDMVPSU, strata = ~SDMVSTRA, weights = ~WTMEC6YR, data = train_data, nest = TRUE)

  # Fit survey-weighted logistic regression
  model <- svyglm(BPXDI1 ~ RIDAGEYR + female + noplace + poor + education_L12 + singlefamily + crowded
    + rented + notemployed, design = train_design)

  preds <- predict(model, newdata = clean_test_data)

  actuals <- clean_test_data$BPXDI1

  # Calculate metrics
  results_df[i, "iteration"] <- i
  results_df[i, "mse_val"] <- mean((actuals - preds)^2)
  results_df[i, "rmse_val"] <- sqrt(mean((actuals - preds)^2))
  results_df[i, "mae_val"] <- mean(abs(actuals - preds))
}
print(mean(results_df$mse_val))

extract_qr_coefs <- function(fit) {
  fit |>
    coef() |>
    t() |>
    as_tibble(rownames = "term") |>
    mutate(term = parse_number(term)) |>
    set_names(
      c(
        "quantiles", "intercept", "age_beta",
        "female_beta", "noplace_beta", "poor_beta",
        "education_L12_beta", "singlefamily_beta",
        "crowded_beta", "rented_beta", "notemployed_beta"
      )
    )
}
