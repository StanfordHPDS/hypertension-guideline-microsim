  margins = c(
    "unif", "binom", "binom",
    "binom", "binom", "gamma",
    "norm", "norm", "binom", "binom",
    "binom", "binom", "binom", "binom",
    "nbinom"

  paramMargins = list(
    # age
    list(min = age_min, max = age_max),
    # female
    list(size = 1, prob = female_prop),
    # black
    list(size = 1, prob = black_prop),
    # place
    list(size = 1, prob = place_prop),
    # insurance
    list(size = 1, prob = insurance_prop),
